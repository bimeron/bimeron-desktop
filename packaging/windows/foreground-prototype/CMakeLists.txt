project(ForegroundPrototype LANGUAGES CXX)

  message(FATAL_ERROR "Windows-only offline prototype")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(foreground-prototype SHARED foreground.cpp callback-context.cpp)
target_compile_features(foreground-prototype PUBLIC cxx_std_17)
target_compile_definitions(foreground-prototype PRIVATE FOREGROUND_EXPORTS)
target_compile_options(foreground-prototype PRIVATE /W4 /WX)
target_link_libraries(foreground-prototype PRIVATE advapi32 msi secur32)
target_link_options(foreground-prototype PRIVATE /DEPENDENTLOADFLAG:0x800)
add_executable(foreground-test foreground-test.cpp)
target_compile_options(foreground-test PRIVATE /W4 /WX)
target_link_libraries(foreground-test PRIVATE foreground-prototype advapi32)
add_test(NAME foreground-worker COMMAND foreground-test)
set_tests_properties(foreground-worker PROPERTIES TIMEOUT 30)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(WIX_ARCH arm64)
else()
  set(WIX_ARCH x64)
endif()
set(FOREGROUND_DLL "$<TARGET_FILE:foreground-prototype>")
configure_file(foreground-sequence.wxs.in foreground-sequence.configured.wxs @ONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/foreground-sequence.wxs"
  INPUT "${CMAKE_CURRENT_BINARY_DIR}/foreground-sequence.configured.wxs")
add_test(NAME foreground-sequence COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/foreground-sequence-test.py"
  "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/foreground-sequence.wxs" "${WIX_ARCH}")
# No install rules and no connection to production CPack. Never ship this DLL.

# Optional explicitly supplied cached tools only; never download or install.
set(WIX_TOOL_DIR "" CACHE PATH "Existing WiX 3 tool directory for offline fixture")
if(WIX_TOOL_DIR)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/fixture.msi"

  add_custom_target(foreground-msi-fixture ALL
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/fixture.msi")
  add_test(NAME foreground-msi-readonly COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/foreground-msi-test.py"
    "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/fixture.msi" "$<TARGET_FILE:foreground-prototype>" "${WIX_ARCH}")
endif()

# Separate observation-only lab. Never part of ALL, install, or production CPack.
option(FOREGROUND_LAB_TRACE "Enable explicit offline LABTRACEONLY artifact target" OFF)
option(FOREGROUND_LAB_TRACE_CONTEXT "Enable separate numeric callback context trace" OFF)
if(FOREGROUND_LAB_TRACE_CONTEXT AND NOT FOREGROUND_LAB_TRACE)
  message(FATAL_ERROR "Context trace requires FOREGROUND_LAB_TRACE")
endif()
if(FOREGROUND_LAB_TRACE)
  if(NOT WIX_TOOL_DIR)
    message(FATAL_ERROR "Lab requires explicitly supplied cached WiX 3 tools")
  endif()
  set(lab_outputs)
  foreach(variant Success CommitFail PostFinalizeFail)
    if(variant STREQUAL "Success")
      set(point 0)
    elseif(variant STREQUAL "CommitFail")
      set(point 1)
    else()
      set(point 2)
    endif()
    add_library(lab-trace-${variant} SHARED EXCLUDE_FROM_ALL lab-trace.cpp)
    target_compile_features(lab-trace-${variant} PRIVATE cxx_std_17)
    target_compile_options(lab-trace-${variant} PRIVATE /W4 /WX)
    target_compile_definitions(lab-trace-${variant} PRIVATE LAB_FAILPOINT=${point})
    target_link_libraries(lab-trace-${variant} PRIVATE advapi32 msi bcrypt)
    if(FOREGROUND_LAB_TRACE_CONTEXT)
      target_sources(lab-trace-${variant} PRIVATE callback-context.cpp lab-context-record.cpp)
      target_compile_definitions(lab-trace-${variant} PRIVATE FOREGROUND_LAB_TRACE_CONTEXT)
      target_link_libraries(lab-trace-${variant} PRIVATE secur32)
    endif()
    target_link_options(lab-trace-${variant} PRIVATE /DEPENDENTLOADFLAG:0x800)
    foreach(kind PRODUCT UPGRADE)
      if(NOT LAB_${variant}_${kind})
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import uuid; print(str(uuid.uuid4()).upper())"
          OUTPUT_VARIABLE guid OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        set(LAB_${variant}_${kind} "${guid}" CACHE INTERNAL "Immutable lab identity")
      endif()
      set(LAB_${kind} "${LAB_${variant}_${kind}}")
    endforeach()
    set(LAB_DLL "$<TARGET_FILE:lab-trace-${variant}>")
    configure_file(lab-trace.wxs.in lab-trace-${variant}.configured.wxs @ONLY)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.wxs"
      INPUT "${CMAKE_CURRENT_BINARY_DIR}/lab-trace-${variant}.configured.wxs")
    set(output "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.msi")
    add_custom_command(OUTPUT "${output}"
      COMMAND "${WIX_TOOL_DIR}/candle.exe" -nologo -out "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.wixobj"
        "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.wxs"
      COMMAND "${WIX_TOOL_DIR}/light.exe" -nologo -v -out "${output}"
        "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.wixobj"
      DEPENDS lab-trace-${variant} "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-trace-${variant}.wxs" VERBATIM)
    list(APPEND lab_outputs "${output}")
  endforeach()
  add_custom_target(foreground-lab-trace DEPENDS ${lab_outputs})
  set(trace_profile historical)
  if(FOREGROUND_LAB_TRACE_CONTEXT)
    set(trace_profile context)
    add_executable(lab-context-record-test EXCLUDE_FROM_ALL lab-context-record-test.cpp
      lab-context-record.cpp callback-context.cpp)
    target_compile_features(lab-context-record-test PRIVATE cxx_std_17)
    target_compile_options(lab-context-record-test PRIVATE /W4 /WX)
    target_link_libraries(lab-context-record-test PRIVATE advapi32 secur32 bcrypt)
    add_test(NAME foreground-context-record COMMAND lab-context-record-test)
  endif()
  add_test(NAME foreground-lab-static COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/lab-trace-check.py" "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>" "${WIX_ARCH}" "${trace_profile}")
endif()

# Access-only ACK experiment. Explicit offline target, never installed or shipped.
option(FOREGROUND_LAB_PROTECTED_ACK "Enable isolated protected ACK MSI fixtures" OFF)
option(FOREGROUND_LAB_PROTECTED_ACK_CHECKPOINT "Enable x64-only offline Pending checkpoint fixture" OFF)
if(FOREGROUND_LAB_PROTECTED_ACK_CHECKPOINT AND NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
  message(FATAL_ERROR "Checkpoint fixture requires explicit native x64 constructor; ARM64 is unsupported")
endif()
if(FOREGROUND_LAB_PROTECTED_ACK OR FOREGROUND_LAB_PROTECTED_ACK_CHECKPOINT)
  if(NOT WIX_TOOL_DIR)
    message(FATAL_ERROR "ACK lab requires explicitly supplied cached WiX 3 tools")
  endif()
  set(ack_outputs)
  set(ack_variants)
  if(FOREGROUND_LAB_PROTECTED_ACK)
    list(APPEND ack_variants Success CommitFail PostFinalizeFail)
  endif()
  if(FOREGROUND_LAB_PROTECTED_ACK_CHECKPOINT)
    list(APPEND ack_variants Checkpoint)
  endif()
  foreach(variant IN LISTS ack_variants)
    if(variant STREQUAL "Success")
      set(point 0)
    elseif(variant STREQUAL "CommitFail")
      set(point 1)
    elseif(variant STREQUAL "PostFinalizeFail")
      set(point 2)
    else()
      set(point 3)
    endif()
    add_library(lab-protected-ack-${variant} SHARED EXCLUDE_FROM_ALL lab-protected-ack.cpp lab-protected-file.cpp)
    target_compile_features(lab-protected-ack-${variant} PRIVATE cxx_std_17)
    target_compile_options(lab-protected-ack-${variant} PRIVATE /W4 /WX)
    target_compile_definitions(lab-protected-ack-${variant} PRIVATE LAB_FAILPOINT=${point})
    target_link_libraries(lab-protected-ack-${variant} PRIVATE advapi32 msi bcrypt ole32 shell32)
    target_link_options(lab-protected-ack-${variant} PRIVATE /DEPENDENTLOADFLAG:0x800)
    foreach(kind PRODUCT UPGRADE)
      if(NOT ACK_${variant}_${kind})
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import uuid; print(str(uuid.uuid4()).upper())"
          OUTPUT_VARIABLE guid OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        set(ACK_${variant}_${kind} "${guid}" CACHE INTERNAL "Immutable ACK lab identity")
      endif()
      set(LAB_${kind} "${ACK_${variant}_${kind}}")
    endforeach()
    set(LAB_DLL "$<TARGET_FILE:lab-protected-ack-${variant}>")
    configure_file(lab-protected-ack.wxs.in lab-protected-ack-${variant}.configured.wxs @ONLY)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.wxs"
      INPUT "${CMAKE_CURRENT_BINARY_DIR}/lab-protected-ack-${variant}.configured.wxs")
    set(output "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.msi")
    add_custom_command(OUTPUT "${output}"
      COMMAND "${WIX_TOOL_DIR}/candle.exe" -nologo -out "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.wixobj"
        "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.wxs"
      COMMAND "${WIX_TOOL_DIR}/light.exe" -nologo -v -out "${output}"
        "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.wixobj"
      DEPENDS lab-protected-ack-${variant} "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-${variant}.wxs" VERBATIM)
    if(variant STREQUAL "Checkpoint")
      add_custom_target(foreground-lab-protected-ack-checkpoint DEPENDS "${output}" lab-protected-ack-checkpoint-test)
    else()
      list(APPEND ack_outputs "${output}")
    endif()
  endforeach()
  if(FOREGROUND_LAB_PROTECTED_ACK)
    add_executable(lab-protected-ack-test EXCLUDE_FROM_ALL lab-protected-ack-test.cpp lab-protected-file.cpp)
    target_compile_features(lab-protected-ack-test PRIVATE cxx_std_17)
    target_compile_options(lab-protected-ack-test PRIVATE /W4 /WX)
    target_compile_definitions(lab-protected-ack-test PRIVATE LAB_PROTECTED_FILE_TESTING)
    target_link_libraries(lab-protected-ack-test PRIVATE advapi32 ole32 shell32 msi bcrypt)
    add_custom_target(foreground-lab-protected-ack DEPENDS ${ack_outputs} lab-protected-ack-test)
    add_test(NAME foreground-lab-protected-ack-native COMMAND lab-protected-ack-test
      "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-Success.run.manifest"
      "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-Success.msi")
    set_tests_properties(foreground-lab-protected-ack-native PROPERTIES DEPENDS foreground-lab-protected-ack-readonly TIMEOUT 30)
    add_test(NAME foreground-lab-protected-ack-readonly COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/lab-protected-ack-check.py" "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>" "${WIX_ARCH}")
  endif()
  if(FOREGROUND_LAB_PROTECTED_ACK_CHECKPOINT)
    add_executable(lab-protected-ack-checkpoint-test EXCLUDE_FROM_ALL lab-protected-ack-test.cpp lab-protected-file.cpp)
    target_compile_features(lab-protected-ack-checkpoint-test PRIVATE cxx_std_17)
    target_compile_options(lab-protected-ack-checkpoint-test PRIVATE /W4 /WX)
    target_compile_definitions(lab-protected-ack-checkpoint-test PRIVATE LAB_PROTECTED_FILE_TESTING LAB_FAILPOINT=3)
    target_link_libraries(lab-protected-ack-checkpoint-test PRIVATE advapi32 ole32 shell32 msi bcrypt)
    add_test(NAME foreground-lab-protected-ack-checkpoint-readonly COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/lab-protected-ack-check.py" "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>" "${WIX_ARCH}" Checkpoint)
    add_test(NAME foreground-lab-protected-ack-checkpoint-model COMMAND lab-protected-ack-checkpoint-test
      "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-Checkpoint.run.manifest"
      "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/lab-protected-ack-Checkpoint.msi")
    set_tests_properties(foreground-lab-protected-ack-checkpoint-model PROPERTIES DEPENDS foreground-lab-protected-ack-checkpoint-readonly TIMEOUT 30)
  endif()
endif()
