cmake_minimum_required(VERSION 3.21)
project(LabCostingProbe LANGUAGES CXX)
add_executable(lab-storage storage.cpp)
target_link_libraries(lab-storage PRIVATE ole32)
add_executable(lab-costing-probe probe.cpp)
target_compile_features(lab-costing-probe PRIVATE cxx_std_17)
target_link_libraries(lab-costing-probe PRIVATE msi ole32 bcrypt)
foreach(t lab-storage lab-costing-probe)
  set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
